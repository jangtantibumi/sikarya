    function drag(ev) {
        ev.dataTransfer.setData("text/plain", ev.currentTarget.id);

    }

    async function drop(ev) {
        ev.preventDefault();
        const box = ev.target.closest('.division-box');
        
        // Reset border styles for all boxes
        document.querySelectorAll('.division-box').forEach(b => {
            b.style.borderColor = b.getAttribute('data-id') ? 'transparent' : 'var(--danger)'; 
        });

        if(!box) return;

        const dataId = ev.dataTransfer.getData("text/plain") || ev.dataTransfer.getData("text");
        const featureKey = ev.dataTransfer.getData("feature_key");
        const divisionId = box.getAttribute('data-id');

        const draggedElement = document.getElementById(dataId);
        const targetList = box.querySelector('.module-list');
        
        if(draggedElement && targetList) {
            targetList.appendChild(draggedElement);
            
            // Check for empty placeholders in all boxes
            updateEmptyPlaceholders();

            // API Call to assign feature
            try {
                const response = await fetch('{{ route('master-demo.features.assign') }}', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-CSRF-TOKEN': '{{ csrf_token() }}'
                    },
                    body: JSON.stringify({
                        feature_key: featureKey,
                        division_id: divisionId || null
                    })
                });
                
                const result = await response.json();
                if(result.success) {
                    window.dispatchEvent(new CustomEvent('toast', { detail: { message: 'Modul berhasil dipindahkan.', type: 'success' }}));
                    
                    // Update sidebar dynamically without reload
                    const sidebarItem = document.getElementById('sidebar-nav-item-' + featureKey);
                    let targetSidebarGroup = document.getElementById('sidebar-nav-group-' + (divisionId || 'uncategorized'));
                    
                    if(sidebarItem && targetSidebarGroup) {
                        targetSidebarGroup.appendChild(sidebarItem);
                    }
                }
            } catch(e) {
                console.error(e);
                window.dispatchEvent(new CustomEvent('toast', { detail: { message: 'Gagal memindahkan modul.', type: 'danger' }}));
            }
        }
    }

    function updateEmptyPlaceholders() {
        document.querySelectorAll('.division-box .module-list').forEach(list => {
            // Remove existing placeholder if any
            const existingPlaceholder = list.querySelector('.empty-placeholder');
            if (existingPlaceholder) existingPlaceholder.remove();

            // If empty, add placeholder
            if (list.children.length === 0) {
                const placeholder = document.createElement('div');
                placeholder.className = 'empty-placeholder';
                placeholder.style = 'border: 2px dashed var(--panel-border); border-radius: 6px; padding: 24px; text-align: center; color: var(--text-muted); font-size: 13px; display: flex; flex-direction: column; align-items: center; justify-content: center; gap: 8px; flex: 1; min-height: 80px;';
                placeholder.innerHTML = '<i class="fa-solid fa-plus" style="font-size: 18px; color: var(--panel-border);"></i> <span>Drop modul ke divisi ini</span>';
                list.appendChild(placeholder);
            }
        });
    }
